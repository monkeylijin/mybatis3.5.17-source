```java
@Override
public boolean apply(DynamicContext context) {
    // 1. 求值test表达式，获取实际值
    Object testValue = evaluateExpression(test, context.getBindings());
    
    // 2. 遍历case分支，找到匹配的
    for (CaseNode caseNode : caseNodes) {
        Object caseValue = evaluateExpression(caseNode.getValue(), context.getBindings());
        
        // 3. 比较值是否相等
        if (Objects.equals(testValue, caseValue)) {
            // 4. 应用匹配的case内容
            caseNode.getContents().apply(context);
            return true;
        }
    }
    
    // 5. 没有匹配的case，应用default
    if (defaultNode != null) {
        defaultNode.apply(context);
        return true;
    }
    
    return false;
}

/**
 * 求值OGNL表达式
 */
private Object evaluateExpression(String expression, Map<String, Object> bindings) {
    try {
        return OgnlCache.getValue(expression, bindings);
    } catch (Exception e) {
        throw new RuntimeException("Failed to evaluate expression: " + expression, e);
    }
}
```
</details>

### 步骤3：实现XML解析器

```java
package com.example.xmltags;

import org.apache.ibatis.parsing.XNode;
import org.apache.ibatis.scripting.xmltags.SqlNode;
import org.apache.ibatis.scripting.xmltags.XMLScriptBuilder;

import java.util.ArrayList;
import java.util.List;

/**
 * Switch标签解析器
 */
public class SwitchNodeHandler {
    
    private final XMLScriptBuilder parser;
    
    public SwitchNodeHandler(XMLScriptBuilder parser) {
        this.parser = parser;
    }
    
    public SqlNode handleNode(XNode nodeToHandle) {
        String test = nodeToHandle.getStringAttribute("test");
        
        // 解析case节点
        List<SwitchSqlNode.CaseNode> caseNodes = new ArrayList<>();
        List<XNode> children = nodeToHandle.getChildren();
        SqlNode defaultNode = null;
        
        for (XNode child : children) {
            String nodeName = child.getName();
            
            if ("case".equals(nodeName)) {
                String value = child.getStringAttribute("value");
                SqlNode contents = parser.parseDynamicTags(child);
                caseNodes.add(new SwitchSqlNode.CaseNode(value, contents));
            } else if ("default".equals(nodeName)) {
                defaultNode = parser.parseDynamicTags(child);
            }
        }
        
        return new SwitchSqlNode(test, caseNodes, defaultNode);
    }
}
```

            .collect(Collectors.joining(","));

### 步骤1：设计缓存键

```java
/**
 * 缓存键生成器
 */
public class CacheKeyGenerator {
    
    /**
     * 生成缓存键
     * 
     * 关键：要反映参数对SQL结构的影响，而不是参数值
     */
    public String generate(MappedStatement ms, Object parameterObject) {
        StringBuilder key = new StringBuilder();
        
        // 1. Statement ID
        key.append(ms.getId());
        key.append(":");
        
        // 2. 参数结构特征
        if (parameterObject == null) {
            key.append("null");
        } else if (parameterObject instanceof Map) {
            Map<?, ?> paramMap = (Map<?, ?>) parameterObject;
            
            // 使用key集合和非null的key作为特征
            key.append("map:");
            key.append(String.join(",", 
                paramMap.keySet().stream()
                    .map(Object::toString)
                    .sorted()
                    .collect(Collectors.toList())
            ));
            
            key.append(":");
            key.append(String.join(",",
                paramMap.entrySet().stream()
                    .filter(e -> e.getValue() != null)
                    .map(e -> e.getKey().toString())
                    .sorted()
                    .collect(Collectors.toList())
            ));
        } else {
            // POJO对象
            // TODO: 实现POJO的缓存键生成
        }
        
        return key.toString();
    }
}
```

### 步骤2：设计缓存值

```java
/**
 * 缓存的SQL信息
 */
@Data
public class CachedSql {
    private final String sql;
    private final List<String> parameterProperties;
    private final List<JdbcType> jdbcTypes;
    private final List<Class<?>> javaTypes;
    
    /**
     * 从BoundSql创建
     */
    public static CachedSql from(BoundSql boundSql) {
        List<ParameterMapping> pms = boundSql.getParameterMappings();
        
        return new CachedSql(
            boundSql.getSql(),
            pms.stream().map(ParameterMapping::getProperty).collect(Collectors.toList()),
            pms.stream().map(ParameterMapping::getJdbcType).collect(Collectors.toList()),
            pms.stream().map(ParameterMapping::getJavaType).collect(Collectors.toList())
        );
    }
    
    /**
     * 转换回BoundSql
     */
    public BoundSql toBoundSql(Configuration configuration, Object parameterObject) {
        // 重建ParameterMapping列表
        List<ParameterMapping> pms = new ArrayList<>(parameterProperties.size());
        for (int i = 0; i < parameterProperties.size(); i++) {
            ParameterMapping pm = new ParameterMapping.Builder(
                configuration,
                parameterProperties.get(i),
                javaTypes.get(i)
            )
            .jdbcType(jdbcTypes.get(i))
            .build();
            
            pms.add(pm);
        }
        
        return new BoundSql(configuration, sql, pms, parameterObject);
    }
}
```

### 步骤3：实现插件

```java
@Intercepts({
    @Signature(
        type = MappedStatement.class,
        method = "getBoundSql",
        args = {Object.class}
    )
})
public class DynamicSqlCachePlugin implements Interceptor {
    
    private final Cache<String, CachedSql> sqlCache;
    private final CacheKeyGenerator keyGenerator;
    
    public DynamicSqlCachePlugin() {
        this.sqlCache = Caffeine.newBuilder()
            .maximumSize(1000)
            .expireAfterWrite(1, TimeUnit.HOURS)
            .recordStats()
            .build();
        this.keyGenerator = new CacheKeyGenerator();
    }
    
    @Override
    public Object intercept(Invocation invocation) throws Throwable {
        MappedStatement ms = (MappedStatement) invocation.getTarget();
        Object parameterObject = invocation.getArgs()[0];
        
        // 1. 只缓存动态SQL
        if (!(ms.getSqlSource() instanceof DynamicSqlSource)) {
            return invocation.proceed();
        }
        
        // 2. 生成缓存键
        String cacheKey = keyGenerator.generate(ms, parameterObject);
        
        // 3. 尝试从缓存获取
        CachedSql cachedSql = sqlCache.getIfPresent(cacheKey);
        if (cachedSql != null) {
            return cachedSql.toBoundSql(ms.getConfiguration(), parameterObject);
        }
        
        // 4. 缓存未命中，执行原方法
        BoundSql boundSql = (BoundSql) invocation.proceed();
        
        // 5. 缓存结果
        sqlCache.put(cacheKey, CachedSql.from(boundSql));
        
        return boundSql;
    }
    
    @Override
    public Object plugin(Object target) {
        return Plugin.wrap(target, this);
    }
    
    @Override
    public void setProperties(Properties properties) {
        // 读取配置...
    }
    
    /**
     * 获取缓存统计
     */
    public CacheStats getStats() {
        return sqlCache.stats();
    }
}
```

@Test
public void testCachePlugin() {
    DynamicSqlCachePlugin plugin = getPlugin();
    
    // 第一次查询（缓存未命中）
    long start1 = System.nanoTime();
    userMapper.searchUsers(query1);
    long time1 = System.nanoTime() - start1;
    
    // 第二次查询（缓存命中）
    long start2 = System.nanoTime();
    userMapper.searchUsers(query2);  // 参数值不同，但结构相同
    long time2 = System.nanoTime() - start2;
    
    // 验证性能提升
    assertTrue(time2 < time1 * 0.5);
    
    // 验证缓存统计
    CacheStats stats = plugin.getStats();
    assertEquals(1, stats.hitCount());
    assertEquals(1, stats.missCount());
}
