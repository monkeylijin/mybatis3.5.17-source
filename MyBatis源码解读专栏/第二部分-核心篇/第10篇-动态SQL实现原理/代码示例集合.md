```xml
            </if>

                    AND name = #{name}
                </when>
                <when test="email != null">
                    AND email = #{email}

    <!-- ========== 示例10：复杂查询（综合应用） ========== -->
    
    <select id="searchUsers" parameterType="com.example.domain.UserQuery" resultType="User">
        <!-- 使用 <bind> 处理模糊查询 -->
        <if test="name != null and name != ''">
            <bind name="namePattern" value="'%' + name + '%'"/>
        </if>
        
        <!-- 计算是否有过滤条件 -->
        <bind name="hasFilters" value="name != null or email != null or status != null"/>
        
        SELECT <include refid="userColumns"/>
        FROM t_user
        <where>
            <!-- 基础条件 -->
            <if test="name != null and name != ''">
                AND name LIKE #{namePattern}
            </if>
            <if test="email != null and email != ''">
                AND email = #{email}
            </if>
            
            <!-- 年龄范围 -->
            <if test="minAge != null and maxAge != null">
                AND age BETWEEN #{minAge} AND #{maxAge}
            </if>
            <if test="minAge != null and maxAge == null">
                AND age >= #{minAge}
            </if>
            <if test="minAge == null and maxAge != null">
                AND age <= #{maxAge}
            </if>
            
            <!-- 状态选择 -->
            <choose>
                <when test="status != null">
                    AND status = #{status}
                </when>
                <otherwise>
                    AND status != 'DELETED'
                </otherwise>
            </choose>
            
            <!-- ID批量查询 -->
            <if test="ids != null and ids.size() > 0">
                AND id IN
                <foreach collection="ids" item="id" open="(" separator="," close=")">
                    #{id}
                </foreach>
            </if>
            
            <!-- 日期范围 -->
            <if test="createdAfter != null">
                AND created_at >= #{createdAfter}
            </if>
            <if test="createdBefore != null">
                AND created_at <= #{createdBefore}
            </if>
        </where>
        
        <!-- 动态排序 -->
        <if test="orderBy != null and orderBy != ''">
            ORDER BY ${orderBy}
        </if>
    </select>
    
</mapper>
```

### 3.3 UserSqlProvider.java

```java
package com.example.provider;

import org.apache.ibatis.jdbc.SQL;
import java.util.List;
import java.util.Map;

/**
 * 编程式SQL构建
 */
public class UserSqlProvider {
    
    /**
     * 示例9：使用 @SelectProvider
     */
    public String findByProvider(Map<String, Object> params) {
        return new SQL() {{
            SELECT("id, name, email, age, status, created_at");
            FROM("t_user");
            
            // 动态添加条件
            if (params.get("name") != null) {
                WHERE("name = #{name}");
            }
            
            if (params.get("minAge") != null) {
                WHERE("age >= #{minAge}");
            }
            
            if (params.get("maxAge") != null) {
                WHERE("age <= #{maxAge}");
            }
            
            if (params.get("status") != null) {
                WHERE("status = #{status}");
            }
            
            // ID列表IN查询
            @SuppressWarnings("unchecked")
            List<Long> ids = (List<Long>) params.get("ids");
            if (ids != null && !ids.isEmpty()) {
                StringBuilder inClause = new StringBuilder("id IN (");
                for (int i = 0; i < ids.size(); i++) {
                    if (i > 0) inClause.append(",");
                    inClause.append("#{ids[").append(i).append("]}");
                }
                inClause.append(")");
                WHERE(inClause.toString());
            }
            
            // 排序
            String orderBy = (String) params.get("orderBy");
            if (orderBy != null && !orderBy.isEmpty()) {
                ORDER_BY(orderBy);
            }
        }}.toString();
    }
}
```

```java
import org.apache.ibatis.io.Resources;
import org.apache.ibatis.mapping.BoundSql;
import org.apache.ibatis.mapping.MappedStatement;
import org.apache.ibatis.mapping.ParameterMapping;
import org.apache.ibatis.session.Configuration;
import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;
import org.apache.ibatis.session.SqlSessionFactoryBuilder;

import java.io.InputStream;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

/**
 * 动态SQL演示
 */
public class DynamicSqlDemo {
    
    private static SqlSessionFactory sqlSessionFactory;
    
    static {
        try {
            InputStream is = Resources.getResourceAsStream("mybatis-config.xml");
            sqlSessionFactory = new SqlSessionFactoryBuilder().build(is);
        } catch (Exception e) {
            throw new RuntimeException("初始化失败", e);
        }
    }
    
    public static void main(String[] args) {
        System.out.println("========== MyBatis 动态SQL演示 ==========\n");
        
        demo1_BasicIf();
        demo2_Where();
        demo3_Foreach();
        demo4_Choose();
        demo5_Bind();
        demo6_Set();
        demo7_Nested();
        demo8_Provider();
        demo9_DebugBoundSql();
        
        System.out.println("\n========== 演示完成 ==========");
    }
    
    /**
     * 示例1：基础 <if> 条件查询
     */
    private static void demo1_BasicIf() {
        System.out.println("========== 示例1：基础 <if> 条件查询 ==========");
        
        try (SqlSession session = sqlSessionFactory.openSession()) {
            UserMapper mapper = session.getMapper(UserMapper.class);
            
            // 场景1：只有name条件
            UserQuery query1 = new UserQuery();
            query1.setName("张三");
            List<User> users1 = mapper.findByCondition(query1);
            System.out.println("场景1（name=张三）：" + users1.size() + " 条记录");
            
            // 场景2：name + status
            UserQuery query2 = new UserQuery();
            query2.setName("张三");
            query2.setStatus("ACTIVE");
            List<User> users2 = mapper.findByCondition(query2);
            System.out.println("场景2（name=张三 AND status=ACTIVE）：" + users2.size() + " 条记录");
        }
        System.out.println();
    }
    
    /**
     * 示例2：<where> 标签使用
     */
    private static void demo2_Where() {
        System.out.println("========== 示例2：<where> 标签使用 ==========");
        
        try (SqlSession session = sqlSessionFactory.openSession()) {
            UserMapper mapper = session.getMapper(UserMapper.class);
            
            UserQuery query = new UserQuery();
            query.setMinAge(25);
            query.setMaxAge(35);
            query.setStatus("ACTIVE");
            
            List<User> users = mapper.findWithWhere(query);
            System.out.println("查询结果：" + users.size() + " 条记录");
            users.forEach(user -> System.out.println("  - " + user.getName() + ", " + user.getAge() + "岁"));
        }
        System.out.println();
    }
    
    /**
     * 示例3：<foreach> 遍历集合
     */
    private static void demo3_Foreach() {
        System.out.println("========== 示例3：<foreach> 遍历集合 ==========");
        
        try (SqlSession session = sqlSessionFactory.openSession()) {

        try (SqlSession session = sqlSessionFactory.openSession()) {
            Configuration configuration = session.getConfiguration();
            
            // 准备参数
            UserQuery query = new UserQuery();
            query.setName("张三");
            query.setMinAge(25);
            query.setIds(Arrays.asList(1L, 2L, 3L));
            
            // 获取 MappedStatement
            String statementId = "com.example.mapper.UserMapper.searchUsers";
            MappedStatement ms = configuration.getMappedStatement(statementId);
            
            // 获取 BoundSql
            BoundSql boundSql = ms.getBoundSql(query);
            
            // 打印生成的SQL
            System.out.println("生成的SQL：");
            System.out.println(boundSql.getSql());
            
            // 打印参数映射
            System.out.println("\n参数映射：");
            List<ParameterMapping> parameterMappings = boundSql.getParameterMappings();
            for (int i = 0; i < parameterMappings.size(); i++) {
                ParameterMapping pm = parameterMappings.get(i);
                String property = pm.getProperty();
                
                // 获取参数值
                Object value;
                if (boundSql.hasAdditionalParameter(property)) {
                    value = boundSql.getAdditionalParameter(property);
                    System.out.println("  [" + i + "] " + property + " = " + value + " (额外参数)");
                } else {
                    System.out.println("  [" + i + "] " + property + " (普通参数)");
                }
            }
        }
        System.out.println();
    }
```

## 7. 性能测试

### 7.1 PerformanceTest.java

```java
package com.example.demo;

import com.example.domain.UserQuery;
import com.example.mapper.UserMapper;
import org.apache.ibatis.io.Resources;
import org.apache.ibatis.session.SqlSession;
import org.apache.ibatis.session.SqlSessionFactory;
import org.apache.ibatis.session.SqlSessionFactoryBuilder;

import java.io.InputStream;

/**
 * 性能测试
 */
public class PerformanceTest {
    
    public static void main(String[] args) throws Exception {
        InputStream is = Resources.getResourceAsStream("mybatis-config.xml");
        SqlSessionFactory factory = new SqlSessionFactoryBuilder().build(is);
        
        System.out.println("========== 动态SQL性能测试 ==========\n");
        
        // 预热
        warmUp(factory);
        
        // 测试动态SQL
        long dynamicTime = testDynamicSql(factory);
        
        System.out.println("\n========== 测试结果 ==========");
        System.out.println("动态SQL平均耗时：" + dynamicTime / 1000 + "μs");
    }
    
    private static void warmUp(SqlSessionFactory factory) {
        try (SqlSession session = factory.openSession()) {
            UserMapper mapper = session.getMapper(UserMapper.class);
            UserQuery query = new UserQuery();
            query.setName("张三");
            
            for (int i = 0; i < 1000; i++) {
                mapper.findByCondition(query);
            }
        }
    }
    
    private static long testDynamicSql(SqlSessionFactory factory) {
        int iterations = 10000;
        long startTime = System.nanoTime();
        
        try (SqlSession session = factory.openSession()) {
            UserMapper mapper = session.getMapper(UserMapper.class);
            
            UserQuery query = new UserQuery();
            query.setName("张三");
            query.setMinAge(25);
            query.setStatus("ACTIVE");
            
            for (int i = 0; i < iterations; i++) {
                mapper.searchUsers(query);
            }
        }
        
        long endTime = System.nanoTime();
        return (endTime - startTime) / iterations;
    }
}
```
